📋 WHAT IT DOES
───────────────
✓ Detects URLs in PDF files
✓ Scrapes web content from those URLs
✓ Vectorizes both PDF and web content
✓ Stores everything in Pinecone with metadata

🚀 BASIC USAGE
──────────────
from app.core.kb.manager import add_file_to_kb

# Automatically scrapes URLs (NEW default behavior)
add_file_to_kb("materials/pdfs/catalog.pdf")

⚙️ ADVANCED USAGE
─────────────────
from app.core.kb.manager import add_file_to_kb_with_urls

# Enable URL scraping (default)
add_file_to_kb_with_urls("file.pdf", scrape_urls=True)

# Disable URL scraping (original behavior)
add_file_to_kb_with_urls("file.pdf", scrape_urls=False)

🔍 EXTRACT URLs ONLY
─────────────────────
from app.core.kb.manager import extract_urls_from_text

urls = extract_urls_from_text("Visit https://example.com")
# Returns: ['https://example.com']

🌐 SCRAPE SINGLE URL
────────────────────
from app.core.kb.manager import scrape_url_content

content = scrape_url_content("https://example.com")
# Returns: "text content from webpage..."

🧪 RUN TESTS
────────────
